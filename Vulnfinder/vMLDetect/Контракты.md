## Контракты входных данных

> [!attention] 
> В проекте используются датасеты с разным форматом исходных данных, для приведения которых к общему виду, используются `контракты`.
> 
> Ниже приведён общий вид `сэмпла`

```json
{
  "sample_id": "...",
  "source_id": "...",
  "source_record_id": "...",
  "raw_label": "1",
  "label_mode": "before_after",
  "code_view": "source",
  "source_code": "...",
  "code_views": {
    "views": ["source_raw", "source_normalized", "source_pseudo_decompiled"],
    "preferred_view": "source_pseudo_decompiled",
    "domain_confidence": "pseudo_decompiled"
  },
  "project_id": "...",
  "repo_url": "...",
  "commit_id": "...",
  "cve_id": "...",
  "cwe_ids": ["CWE-120"],
  "function_name": "...",
  "file_path": "...",
  "line_spans": ["12:14"],
  "before_after_pair_id": "...",
  "pair_role": "before",
  "language": "C/C++",
  "metadata": {}
}
```
